\begin{frame}{Общая постановка}
Ищется минимум гладкой функции:
\[
    \min_{x\in\mathbb{R}^n} f(x), \qquad \norm{\grad f(x_k)} \le \eps.
\]

\begin{block}{Базовая схема спуска}
\[
    x_{k+1}=x_k+\alpha_k p_k,
    \qquad p_k \approx -\grad f(x_k).
\]
\end{block}

\begin{itemize}
    \item В ЛР2 направление фиксировано как антиградиент, исследуется выбор $\alpha_k$.
    \item В ЛР3 направление и масштаб шага меняются за счёт памяти метода.
    \item Во всех экспериментах ЛР3 использовалась точность $\eps=10^{-8}$.
\end{itemize}
\end{frame}

\begin{frame}{Тестовые функции}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\textbf{Квадратичные функции}
\[
    f(x)=\frac12 x^T A x,
    \qquad \grad f(x)=Ax.
\]
\begin{itemize}
    \item Хорошо обусловленная: близкая кривизна по координатам.
    \item Плохо обусловленная: узкая вытянутая долина.
\end{itemize}

\column{0.48\textwidth}
\textbf{Сложные функции}
\begin{itemize}
    \item Rosenbrock: узкая изогнутая долина, минимум $(1,1)$.
    \item Ackley: локальные колебания, глобальный минимум $(0,0)$.
    \item Himmelblau: четыре глобальных минимума и области притяжения.
\end{itemize}
\end{columns}

\smallnote{Квадратичные функции показывают влияние обусловленности, сложные функции проверяют устойчивость траекторий и критериев остановки.}
\end{frame}

\begin{frame}{ЛР2: классические методы выбора шага}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Методы}
\begin{itemize}
    \item ConstantStepGD: $\alpha_k=\alpha$.
    \item ArmijoBacktracking: дробление шага до достаточного убывания.
    \item StrongWolfe: достаточное убывание и контроль кривизны.
    \item SteepestDescent: одномерная минимизация вдоль антиградиента.
\end{itemize}

\column{0.5\textwidth}
\textbf{Условие Армихо}
\[
 f(x_k+\alpha p_k) \le f(x_k)+c_1\alpha\grad f(x_k)^Tp_k.
\]
\textbf{Сильное условие Вольфе}
\[
 |\grad f(x_k+\alpha p_k)^Tp_k| \le c_2 |\grad f(x_k)^Tp_k|.
\]
\end{columns}
\end{frame}

\begin{frame}{ЛР3: идея модификаций градиентного спуска}
\begin{block}{Общая форма}
\[
    x_{k+1}=x_k-\alpha_k d_k,
\]
где $d_k$ строится не только по текущему градиенту, но и по внутреннему состоянию метода.
\end{block}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{Инерционные методы}
\begin{itemize}
    \item Momentum;
    \item Nesterov.
\end{itemize}
\column{0.48\textwidth}
\textbf{Адаптивные методы}
\begin{itemize}
    \item AdaGrad;
    \item RMSProp;
    \item AdaDelta;
    \item Adam.
\end{itemize}
\end{columns}

\smallnote{Из лекций: модификации можно понимать как фильтрацию направления и координатное масштабирование шага.}
\end{frame}

\begin{frame}{ЛР3: основные формулы методов}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\textbf{Momentum}
\[
    v_{k+1}=\beta v_k+\grad f(x_k),\qquad x_{k+1}=x_k-\alpha v_{k+1}.
\]
\textbf{AdaGrad}
\[
    s_{k+1}=s_k+g_k^2,\qquad x_{k+1}=x_k-\alpha\frac{g_k}{\sqrt{s_{k+1}}+\delta}.
\]
\column{0.49\textwidth}
\textbf{RMSProp}
\[
    s_{k+1}=\rho s_k+(1-\rho)g_k^2.
\]
\textbf{Adam}
\[
    m_{k+1}=\beta_1m_k+(1-\beta_1)g_k,
\]
\[
    v_{k+1}=\beta_2v_k+(1-\beta_2)g_k^2.
\]
\end{columns}
\smallnote{Все операции с квадратами и делением в адаптивных методах выполняются покоординатно.}
\end{frame}

\begin{frame}{Сравнение ЛР2 и ЛР3}
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\textbf{ЛР2: линейный поиск}
\begin{itemize}
    \item меняется длина шага $\alpha_k$;
    \item направление остаётся антиградиентным;
    \item сильная сторона: контролируемое убывание функции;
    \item цена: дополнительные вызовы $f$ и $\grad f$.
\end{itemize}

\column{0.48\textwidth}
\textbf{ЛР3: модификации спуска}
\begin{itemize}
    \item меняются направление и масштаб шага;
    \item используется внутреннее состояние метода;
    \item сильная сторона: ускорение и адаптация;
    \item риск: чувствительность к гиперпараметрам.
\end{itemize}
\end{columns}

\vspace{0.6em}
\begin{block}{Главная практическая идея}
Линейный поиск делает шаг безопаснее, а модификации градиентного спуска делают направление и масштаб шага информативнее.
\end{block}
\end{frame}

\begin{frame}{Выводы}
\begin{itemize}
    \item Постоянный шаг требует аккуратного выбора: слишком малый шаг медленный, слишком большой может разрушить сходимость.
    \item Armijo, Strong Wolfe и наискорейший спуск уменьшают риск неудачного шага, но увеличивают стоимость итерации.
    \item Momentum и Nesterov ускоряют движение за счёт памяти первого порядка.
    \item RMSProp и Adam оказались более устойчивыми среди адаптивных методов на выбранных сетках параметров.
    \item AdaGrad и AdaDelta нужно оценивать по конечному $f(x_k)$ и $\norm{\grad f(x_k)}$, если все запуски достигли \texttt{max\_iter}.
    \item На Ackley и Himmelblau критерий по градиенту нужно интерпретировать осторожно: стационарная точка не всегда равна нужному глобальному минимуму.
\end{itemize}
\end{frame}
